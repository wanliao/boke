### 3.编写函数
#### 1.判断素数
```c++
int su(int n) {
    // 1. 处理小于等于 1 的情况
    if (n <= 1) return 0;

    // 2. 2 是唯一的偶素数
    if (n == 2) return 1;

    // 3. 排除掉所有的偶数（提高效率）
    if (n % 2 == 0) return 0;

    // 4. 只检查奇数因子，且范围缩小到 sqrt(n)
    int a = sqrt(n);
    for (int i = 3; i <= a; i += 2) {
        if (n % i == 0) {
            return 0; // 找到能整除的因子，说明不是素数
        }
    }

    return 1; // 循环结束未发现因子，则是素数
}
```
